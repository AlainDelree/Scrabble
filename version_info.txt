  ),

    StringFileInfo([
      StringTable(

         StringStruct(u'ProductName', u'Scrabble'),
         StringStruct(u'ProductVersion', u'BUILD.0.0.0')])
    ]),
    VarFileInfo([VarStruct(u'Translation', [0x040C, 1200])])
  ]
)
